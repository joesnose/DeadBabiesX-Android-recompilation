cmake_minimum_required(VERSION 3.20)
project(DeadBabiesXAndroid LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/SDL2-2.32.10" CACHE PATH "SDL 2.32.10 source")
add_subdirectory("${SDL_SOURCE_DIR}" sdl)
set(DBX_SOURCES src/main.cpp src/game.cpp src/frontend.cpp src/xengine.cpp src/selftest.cpp src/vendor/stb_vorbis.c)
if(ANDROID)
  add_library(main SHARED ${DBX_SOURCES})
  target_link_libraries(main PRIVATE SDL2 GLESv1_CM log android)
  target_compile_options(main PRIVATE -O2 -Wno-narrowing -Wno-dangling-else)
else()
  find_path(GLES_INCLUDE_DIR GLES/gl.h REQUIRED)
  find_library(GLES_LIBRARY GLESv1_CM REQUIRED)
  add_executable(dbx-host ${DBX_SOURCES})
  target_compile_definitions(dbx-host PRIVATE DBX_GLES)
  target_include_directories(dbx-host PRIVATE "${GLES_INCLUDE_DIR}")
  target_link_libraries(dbx-host PRIVATE SDL2 "${GLES_LIBRARY}")
endif()
